1"""What a CIF or STAR document holds, in the terms Croissant needs. 

2 

3CIF and STAR share one grammar: a document is a sequence of data blocks, and a 

4block holds tag/value pairs, loops, or both. gemmi parses that grammar; this 

5module says what the parse means for a dataset description, so the handlers for 

6the several formats built on it agree on blocks, columns and types rather than 

7each inventing an answer. 

8 

9Nothing here knows about Croissant nodes. It returns :class:`Table` 

10descriptions; a handler turns those into record sets. 

11""" 

12 

13from __future__ import annotations 

14 

15import re 

16from dataclasses import dataclass 

17from typing import Iterable, List, Tuple 

18 

19try: 

20 import gemmi 

21except ImportError: 

22 # gemmi is an optional extra, so a plain install reaches these handlers 

23 # without it. The three modules that parse through it guard their import 

24 # the same way and refuse their files through :func:`require_gemmi`. 

25 gemmi = None 

26 

27#: How to get gemmi, for the refusal a user without it reads. 

28INSTALL_HINT = 'pip install "croissant-baker[structural-biology]"' 

29 

30#: Rows read per column when inferring its type. A refined particle stack has 

31#: millions of rows and every one of them costs a Python-level call, so the head 

32#: of the column decides: a value below the bound that contradicts the rest is 

33#: not seen, and the column keeps the type its head implies. 

34MAX_SAMPLE_ROWS = 1000 

35 

36#: The two shapes a block can take. A block holding both yields one table of 

37#: each, because a pair is one value for the block and a loop row is one record. 

38KIND_PAIRS = "pairs" 

39KIND_LOOP = "loop" 

40 

41#: What a block with no name after ``data_`` is called, so a table always has a 

42#: name an identifier can be derived from. 

43UNNAMED_BLOCK = "block" 

44 

45INTEGER = "sc:Integer" 

46FLOAT = "sc:Float" 

47TEXT = "sc:Text" 

48 

49# Deliberately narrower than int() and float(), which accept spellings CIF does 

50# not have: int() reads "1_000" as a thousand, and float() reads "nan" and "inf" 

51# as numbers where a file means those literally. 

52_INTEGER = re.compile(r"[+-]?\d+$") 

53_FLOAT = re.compile(r"[+-]?(?:\d+\.?\d*|\.\d+)(?:[eE][+-]?\d+)?$")

56def require_gemmi(module: object, what: str) -> None: 

57 """Refuse a file this install has no parser for, and say what to install. 

58 

59 Args: 

60 module: The caller's own ``gemmi``, which is ``None`` when the extra is 

61 not installed. Passed in rather than read from here, so each module 

62 answers for the import it actually made. 

63 what: The formats the caller reads, named as a user would name them. 

64 

65 Raises: 

66 ValueError: When gemmi is missing. A ValueError rather than an 

67 ImportError because the scan turns it into a refusal for that one 

68 file, so the rest of the bake still runs. 

69 """ 

70 if module is None: 

71 raise ValueError( 

72 f"Reading {what} files needs gemmi, which is not installed. " 

73 f"Install it with: {INSTALL_HINT}" 

74 )

77@dataclass(frozen=True) 

78class Table: 

79 """One table a document holds: a loop, or a block's pairs taken together. 

80 

81 Attributes: 

82 block: The name of the data block it came from, without ``data_``. 

83 name: A name unique within the document, so an identifier derived from 

84 it stays unique too. 

85 kind: :data:`KIND_LOOP` or :data:`KIND_PAIRS`. 

86 columns: ``(column name, Croissant type)`` in the order the file 

87 declares them. 

88 rows: The loop's length, or 1 for a block's pairs. 

89 """ 

90 

91 block: str 

92 name: str 

93 kind: str 

94 columns: Tuple[Tuple[str, str], ...] 

95 rows: int

98def describe(document) -> List[Table]: 

99 """Every table in a parsed document, in file order. 

100 

101 Args: 

102 document: A ``gemmi.cif.Document``, from ``gemmi.cif.read_string`` or 

103 any of gemmi's other readers. 

104 

105 Returns: 

106 One :class:`Table` per loop, plus one per block that carries pairs. A 

107 block with no items at all yields nothing: there is no column to 

108 describe, and a record set without fields is one mlcroissant refuses. 

109 """ 

110 tables: List[Table] = [] 

111 used: set = set() 

112 for block in document: 

113 found = _tables_of(block) 

114 for name, (kind, columns, rows) in zip(_names(block.name, len(found)), found): 

115 tables.append( 

116 Table( 

117 block=block.name, 

118 name=_unique(name, used), 

119 kind=kind, 

120 columns=columns, 

121 rows=rows, 

122 ) 

123 ) 

124 return tables

127def _tables_of(block) -> list: 

128 """``(kind, columns, rows)`` per table in one block, pairs first.""" 

129 pairs = [] 

130 loops = [] 

131 for item in block: 

132 if item.loop is not None: 

133 loops.append(item.loop) 

134 elif item.pair is not None: 134 ↛ 131line 134 didn't jump to line 131 because the condition on line 134 was always true

135 pairs.append(item.pair) 

136 

137 found = [] 

138 if pairs: 

139 columns = tuple( 

140 (_column_name(tag), _column_type([value])) for tag, value in pairs 

141 ) 

142 found.append((KIND_PAIRS, columns, 1)) 

143 for loop in loops: 

144 found.append((KIND_LOOP, _loop_columns(loop), loop.length())) 

145 return found

148def _loop_columns(loop) -> Tuple[Tuple[str, str], ...]: 

149 """One ``(name, type)`` per loop tag, typed from the head of its column.""" 

150 sampled = min(loop.length(), MAX_SAMPLE_ROWS) 

151 return tuple( 

152 ( 

153 _column_name(tag), 

154 _column_type(loop[row, column] for row in range(sampled)), 

155 ) 

156 for column, tag in enumerate(loop.tags) 

157 )

160def _column_name(tag: str) -> str: 

161 """``_rlnImageName`` becomes ``rlnImageName``. 

162 

163 Only the leading underscore goes. A dot inside the tag separates an mmCIF 

164 category from its keyword and belongs to the name: ``_struct.title`` names 

165 one column, not two. 

166 """ 

167 return tag[1:] if tag.startswith("_") else tag

170def _column_type(values: Iterable[str]) -> str: 

171 """The Croissant type every value in the column fits. 

172 

173 The null tokens ``.`` (inapplicable) and ``?`` (unknown) say nothing about 

174 the type and are skipped, so one missing measurement does not turn a column 

175 of floats into text. A column of nothing but nulls is text, which claims the 

176 least. 

177 """ 

178 seen = False 

179 integral = True 

180 for raw in values: 

181 if gemmi.cif.is_null(raw): 

182 continue 

183 value = gemmi.cif.as_string(raw) 

184 seen = True 

185 if not _FLOAT.match(value): 

186 return TEXT 

187 if integral and not _INTEGER.match(value): 

188 integral = False 

189 if not seen: 

190 return TEXT 

191 return INTEGER if integral else FLOAT

194def _names(block: str, count: int) -> List[str]: 

195 """Names for one block's tables, indexed only where there is a choice.""" 

196 base = block or UNNAMED_BLOCK 

197 if count == 1: 

198 return [base] 

199 return [f"{base}_{i}" for i in range(1, count + 1)]

202def _unique(name: str, used: set) -> str: 

203 """``name``, or the first free ``name__N``. 

204 

205 A document may repeat a block name; the identifiers a handler derives from 

206 these names may not repeat, so the collision is settled once, here. 

207 """ 

208 candidate = name 

209 n = 2 

210 while candidate in used: 

211 candidate = f"{name}__{n}" 

212 n += 1 

213 used.add(candidate) 

214 return candidate
