49# conformsTo URIs declared on the Dataset. mlcroissant defaults conforms_to to 

50# 1.0 even on 1.1.x — passing CROISSANT_CONFORMS_TO explicitly is the single 

51# source of truth for our declared spec version. RAI_CONFORMS_TO is appended 

52# to the conformsTo array by _ensure_rai_conforms_to() when RAI fields are 

53# present (the RAI extension vocab itself did NOT version-bump in Croissant 1.1). 

54# https://docs.mlcommons.org/croissant/docs/croissant-spec-1.1.html 

55CROISSANT_CONFORMS_TO = "http://mlcommons.org/croissant/1.1" 

56RAI_CONFORMS_TO = "http://mlcommons.org/croissant/RAI/1.0" 

57BIOSCHEMAS_CONFORMS_TO = "https://bioschemas.org/profiles/Dataset/1.0-RELEASE"

59# Profiles a document can additionally declare, by the name --profile takes. 

60# A new profile is two entries: the URI declared alongside CROISSANT_CONFORMS_TO 

61# here, and the fields the profile requires in PROFILE_MINIMUM_KEYS below. 

62PROFILE_CONFORMS_TO = { 

63 "bioschemas": BIOSCHEMAS_CONFORMS_TO, 

64} 

65 

66# Emitted keys a document must carry before it may declare each profile. 

67# Declaring one is a claim a SHACL validator will check, so a document that 

68# names a profile and then omits what the profile lists as minimum scores 

69# worse than one that declares nothing. The generator refuses instead. 

70# 

71# The list mirrors the profile's own minimums rather than what a bake happens 

72# to be able to omit. Bioschemas Dataset 1.0-RELEASE lists ten; @context, 

73# @type and name are always written and dct:conformsTo is what the declaration 

74# itself adds, so the six below are what is left. description and license are 

75# defaulted today and so can never be reported missing; they stay listed so the 

76# check follows the profile rather than the defaults. 

77# https://bioschemas.org/profiles/Dataset/1.0-RELEASE 

78PROFILE_MINIMUM_KEYS = { 

79 "bioschemas": ("@id", "description", "identifier", "keywords", "license", "url"), 

80}

83def url_has_whitespace(url: str) -> bool: 

84 """Whether ``url`` carries whitespace, which no IRI may. 

85 

86 The document's ``@id`` is an IRI: a url with a space in it makes the whole 

87 document unreadable to a JSON-LD parser, so a url like that is left out of 

88 the ``@id`` rather than written into it. Whitespace is all this asks 

89 about, because it is the one thing that breaks parsing outright, and 

90 ``url`` itself is written as given whatever the answer is. 

91 

92 One owner for the rule: the generator asks before injecting the key, and 

93 the CLI asks before telling the user why the key is missing. Both ask only 

94 when there is a url to ask about. 

95 """ 

96 return any(character.isspace() for character in url)

102 """The profile names to declare, cleaned and checked, or None. 

103 

104 The single owner of the rule: the generator calls it from ``__init__`` and 

105 the CLI calls it from the ``--profile`` callback, so a library caller and a 

106 command line get the same contract and the same message. Strips each name, 

107 drops empties, splits comma lists the way ``--keywords`` and 

108 ``--identifier`` take them, and deduplicates with ``dict.fromkeys`` so a 

109 name given twice is declared once in the order it was first asked for. 

110 

111 A bare string is one flag's worth of input rather than an iterable of 

112 characters, so ``profiles="bioschemas"`` means what it reads as. 

113 

114 Raises: 

115 ValueError: If a name has no conformsTo URI in ``PROFILE_CONFORMS_TO``. 

116 """ 

117 if profiles is None: 

118 return None 

119 if isinstance(profiles, str): 

120 profiles = [profiles] 

121 names = [ 

122 stripped 

123 for value in profiles 

124 for part in value.split(",") 

125 if (stripped := part.strip()) 

126 ] 

127 names = list(dict.fromkeys(names)) 

128 unknown = [name for name in names if name not in PROFILE_CONFORMS_TO] 

129 if unknown: 

130 raise ValueError( 

131 f"Unknown profile(s): {', '.join(repr(name) for name in unknown)}. " 

132 f"Known profiles: {', '.join(sorted(PROFILE_CONFORMS_TO))}" 

133 )

570 # A duplicate stands on its primary's description. Where there is none, 

571 # saying so beats a link to a file nothing describes. 

572 for entry in linked: 

573 primary = entry.duplicate_of 

574 if primary is not None and primary.outcome is Outcome.DESCRIBED: 574 ↛ 576line 574 didn't jump to line 576 because the condition on line 574 was always true

575 continue 

576 staged.pop(entry, None) 

577 target = primary.path if primary is not None else "another file" 

578 entry.failed( 

579 Reason.BUILD_FAILED, 

580 ValueError(f"duplicates {target}, which was not described"), 

581 ) 

582 

583 surviving = [e for e in with_objects if e.outcome is not Outcome.FAILED] 

584 if not any(e.outcome is Outcome.DESCRIBED for e in surviving): 

585 raise ValueError("No supported files found in the dataset") 

586 

587 # Reconciled here, where the staged FileObjects are final, so coverage 

588 # counts what was written rather than what was claimed. 

589 by_scanned_path = {str(e.path): e for e in entries} 

590 for parent in surviving: 

591 _own, *siblings = staged[parent] 

592 for obj in siblings: 

593 sibling = by_scanned_path.get(obj.content_url) 

594 if sibling is not None and sibling.outcome in _CARRIED_BY_ANOTHER: 594 ↛ 592line 594 didn't jump to line 592 because the condition on line 594 was always true

595 sibling.referenced(parent) 

596 

597 # distributions holds both FileObjects and FileSets — the full contents 

598 # of the Croissant `distribution` array per the spec. 

599 distributions = [] 

600 by_stored_path: dict = {} 

601 for entry in surviving: 

602 objects = staged.get(entry) 

603 if objects is None: 603 ↛ 604line 603 didn't jump to line 604 because the condition on line 603 was never true

604 continue 

605 distributions.extend(objects) 

606 by_stored_path[str(entry.path)] = objects[0].id 

607 

608 # A duplicate's sameAs target may not have been built yet: discovery 

609 # order is the filesystem's. Resolve once every id is assigned. 

610 for entry in surviving: 

611 if entry.outcome is Outcome.LINKED and entry in staged: 

612 staged[entry][0].same_as = [ 

613 by_stored_path[str(entry.duplicate_of.path)] 

614 ] 

615 

616 distributions.extend(file_sets) 

617 

618 # A stem shared across two formats only collides here, where every 

619 # batch is visible at once. 

620 record_sets = _disambiguate_record_sets(batches) 

621 

622 if self.detect_references: 

623 self._apply_reference_detection(record_sets) 

624 

625 described_metas = [ 

626 (e.handler, e.meta) for e in entries if e.outcome is Outcome.DESCRIBED 

627 ] 

628 metadata = mlc.Metadata( 

629 name=self.name or self.dataset_path.name, 

630 description=self._build_description(described_metas), 

631 url=self.url, 

632 license=self._resolve_license(), 

633 creators=self._build_creators(), 

634 date_published=self._resolve_date(), 

635 date_created=self._parse_iso(self.date_created), 

636 date_modified=self._parse_iso(self.date_modified), 

637 version=self.version or "1.0.0", 

638 cite_as=self._build_citation(), 

639 conforms_to=self._resolve_conforms_to(), 

640 keywords=self.keywords, 

641 in_language=self.in_language, 

642 same_as=self.same_as, 

643 publisher=self._build_publisher(), 

644 sd_licence=self.sd_license, 

645 **self.rai_fields, 

646 ) 

647 

648 _assert_unique_node_ids(distributions, record_sets) 

649 

650 metadata.distribution = distributions 

651 metadata.record_sets = record_sets 

652 

653 result = metadata.to_json() 

654 # Spec fields without a native mlcroissant parameter — inject 

655 # post-serialisation. Keep keys absent (not null) when the caller 

656 # didn't supply a value, so optional fields don't pollute outputs 

657 # that don't need them. ``sd_version`` IS a native mlc 1.1.0 param, 

658 # but mlc emits it as ``cr:sdVersion`` (no @context alias); the 

659 # canonical 1.1 examples use the unprefixed form, so we write the 

660 # canonical key directly. 

661 if self.sd_version is not None: 

662 result["sdVersion"] = self.sd_version 

663 # mlcroissant takes an ``id`` for the Metadata node but writes no 

664 # top-level @id, leaving the Dataset a blank node that nothing can 

665 # refer to. The dataset URL is the identifier a reader already has, 

666 # and it is what Bioschemas expects there. A url no IRI can be built 

667 # from is skipped rather than written: a declared profile that 

668 # requires @id refuses the bake outright, and the CLI repeats the 

669 # warning below on stderr, where a terminal user will see it. 

670 # Written beside the other node keywords, where a reader looks for the 

671 # subject of the graph, rather than trailing the record sets. 

672 if self.url and not url_has_whitespace(self.url): 

673 result = { 

674 "@context": result.pop("@context"), 

675 "@type": result.pop("@type"), 

676 "@id": self.url, 

677 **result, 

678 } 

679 elif self.url: 

680 logger.warning( 

681 "url %r contains whitespace, so no @id was emitted for the " 

682 "dataset. Percent-encode the whitespace (a space becomes " 

683 "%%20) to give the document an identifier.", 

684 self.url, 

685 ) 

686 if self.alternate_name is not None: 

687 result["alternateName"] = self.alternate_name 

688 if self.is_live_dataset is not None: 

689 result["isLiveDataset"] = self.is_live_dataset 

690 if self.temporal_coverage is not None: 

691 result["temporalCoverage"] = self.temporal_coverage 

692 if self.usage_info is not None: 

693 result["usageInfo"] = self.usage_info 

694 if self.identifier: 

695 # One accession reads as a string, the way mlcroissant flattens its 

696 # own single-element lists; several stay a list. Deduplicated in 

697 # declared order first, so naming an accession twice does not turn 

698 # the same claim into a list of two. 

699 accessions = list(dict.fromkeys(self.identifier)) 

700 result["identifier"] = accessions[0] if len(accessions) == 1 else accessions 

701 if self.conditions_of_access is not None: 

702 result["conditionsOfAccess"] = self.conditions_of_access 

703 if self.is_accessible_for_free is not None: 

704 result["isAccessibleForFree"] = self.is_accessible_for_free 

705 if self.included_in_data_catalog is not None: 

706 # schema.org gives includedInDataCatalog exactly one range, 

707 # DataCatalog. A bare string would be read as a literal under 

708 # @vocab, so the URL rides on a node, the shape publisher 

709 # already uses for its Organization. 

710 result["includedInDataCatalog"] = { 

711 "@type": "sc:DataCatalog", 

712 "url": self.included_in_data_catalog, 

713 } 

714 if self.field_mappings: 

715 _apply_field_mappings(result, self.field_mappings) 

716 self._assert_profile_minimums(result) 

717 return result

719 def _assert_profile_minimums(self, document: dict) -> None: 

720 """Refuse to declare a profile whose minimum fields are missing. 

721 

722 Read off the assembled document rather than the constructor 

723 arguments, because what reaches the file is not always what was 

724 passed: ``description`` is generated when none was given, ``license`` 

725 is defaulted, and ``@id`` is injected from ``url`` when the url can be 

726 an IRI. The check has to agree with what a validator will read. 

727 

728 Raises: 

729 ValueError: naming every missing field, so one run tells the 

730 caller everything they have to supply. 

731 """ 

732 for profile in self.profiles or []: 

733 missing = [ 

734 key 

735 for key in PROFILE_MINIMUM_KEYS.get(profile, ()) 

736 if not document.get(key) 

737 ] 

738 if missing: 

739 raise ValueError( 

740 f"Profile '{profile}' requires fields this document does " 

741 f"not carry: {', '.join(missing)}. They are minimum fields " 

742 "of the profile, so declaring it without them leaves the " 

743 "document failing validation against what it claims. " 

744 "Supply them, or drop the profile." 

745 )

751 def _resolve_worker_count(self, n_files: int) -> int: 

752 """Decide how many threads to use for extraction. 

753 

754 An explicit ``max_workers`` wins (floored at 1). Otherwise auto-size: 1 

755 for trivial inputs, else a modest oversubscription of the CPU count (the 

756 work is I/O-bound — whole-file hashing and header reads — so threads 

757 spend most of their time off-CPU), capped to bound open file descriptors. 

758 """ 

759 if self.max_workers is not None: 

760 return max(1, self.max_workers) 

761 if n_files <= 1: 

762 return 1 

763 cpu = os.cpu_count() or 1 

764 return min(8, n_files, cpu * 2)

766 def describe_refusal(self, entry: ScanEntry) -> str: 

767 """One line for a file that was passed over: the reason, then the file. 

768 

769 Reason first because that is what a reader scans a column of these for; 

770 the path last because it is the long part, and it is where a reader 

771 goes once the reason has told them whether to care. 

772 

773 The path is ``entry.path``: relative to the dataset root, so the line a 

774 reader is shown is the key they can look the file up by in ``--report``. 

775 """ 

776 return f"{entry.detail}. File: {entry.path}"

778 def _extract_entry(self, entry: ScanEntry) -> None: 

779 """Select a handler for one scan entry and resolve its outcome in place. 

780 

781 Touches only that entry, so it is safe to call concurrently. Never 

782 raises: selection is inside the guard as well as extraction, because a 

783 handler sniffing magic bytes has to decompress to do it, so a corrupt 

784 wrapper raises while the registry is still deciding who owns the file. 

785 """ 

786 full_path = self.dataset_path / entry.path 

787 try: 

788 selection = self.handlers.select(full_path, entry.path) 

789 except Exception as e: # noqa: BLE001 — one file, not the bake 

790 entry.failed(Reason.CLAIM_FAILED, e) 

791 return 

792 

793 if selection.handler is None: 

794 entry.unclaimed(selection.reason or Reason.NO_HANDLER, selection.refusal) 

795 return 

796 

797 handler, source = selection.handler, selection.source 

798 try: 

799 meta = extract_with(handler, source, full_path, **self._handler_kwargs) 

800 # The logical path derives identifiers; the stored name is for 

801 # prose, which has to name a file the reader can find on disk. 

802 meta["relative_path"] = str(source.relative_path) 

803 meta["stored_name"] = entry.path.name 

804 entry.ready(handler, meta) 

805 except Exception as e: # noqa: BLE001 — recorded, then reported 

806 entry.failed(Reason.EXTRACT_FAILED, e)

808 def _file_objects_for(self, entry: ScanEntry, counter: int) -> tuple[list, int]: 

809 """Build the distribution entries for one file, and the next free id. 

810 

811 A list, because a multi-file record produces several: WFDB reads a 

812 header together with its sibling ``.dat`` and ``.atr``. Everything here 

813 addresses the file *as stored*, wrapper included. 

814 

815 A handler's ``description`` is how what it could not describe reaches 

816 a reader holding the manifest alone. It is optional and belongs to the 

817 handler: a format whose content is not a table has properties worth 

818 stating and no record set to state them on, and BAM is one. Absent, the 

819 FileObject carries none, which is what most handlers produce today. 

820 """ 

821 meta = entry.meta 

822 objects = [ 

823 mlc.FileObject( 

824 id=f"file_{counter}", 

825 name=entry.path.name, 

826 content_url=str(entry.path), 

827 description=meta.get("description"), 

828 encoding_formats=_encoding_formats( 

829 meta["encoding_format"], entry.path.name 

830 ), 

831 content_size=str(meta["file_size"]), 

832 sha256=meta["sha256"], 

833 ) 

834 ] 

835 counter += 1 

836 

837 for related in meta.get("related_files", []): 

838 rel_path = Path(related["path"]) 

839 objects.append( 

840 mlc.FileObject( 

841 id=f"file_{counter}", 

842 name=related["name"], 

843 content_url=str(rel_path.relative_to(self.dataset_path)), 

844 encoding_formats=[related["encoding"]], 

845 content_size=str(related["size"]), 

846 sha256=related["sha256"], 

847 ) 

848 ) 

849 counter += 1 

850 

851 return objects, counter

853 def _apply_reference_detection(self, record_sets: list) -> None: 

854 """Detect and attach foreign-key cr:references across RecordSets. 

855 

856 Opt-in via ``detect_references``. Delegates the (pure) detection to 

857 ``references.detect_foreign_keys`` and maps its result back onto the real 

858 Field objects. Conservative: a child field's ``references`` is set only 

859 when a shared key column has a parent RecordSet identifiable by name. 

860 

861 What it found is stored on :attr:`reference_report` rather than logged. 

862 Nothing configures logging for this package — ``__init__`` attaches a 

863 NullHandler — so a log record here reaches nobody, and the CLI owns 

864 terminal output. 

865 """ 

866 descriptors = [ 

867 { 

868 "id": rs.id, 

869 "name": rs.name or rs.id, 

870 "columns": [f.name for f in (rs.fields or [])], 

871 } 

872 for rs in record_sets 

873 ] 

874 links, unresolved = detect_foreign_keys(descriptors) 

875 self._reference_report = ReferenceReport(links=links, unresolved=unresolved) 

876 

877 rs_by_id = {rs.id: rs for rs in record_sets} 

878 for link in links: 

879 parent_field = _field_named( 

880 rs_by_id[link["parent_rs"]], link["parent_column"] 

881 ) 

882 child_field = _field_named(rs_by_id[link["child_rs"]], link["column"]) 

883 child_field.references = mlc.Source(field=parent_field.id)

951 def _build_citation(self) -> Optional[str]: 

952 """Build the default citation, or None. 

953 

954 A caller-supplied ``citation`` is used verbatim. Otherwise it is derived 

955 only from real metadata — the supplied creators and the year of the 

956 publication/creation date — so the same input always bakes to the same 

957 citation (no wall-clock state). With neither a creator nor a date there 

958 is nothing real to cite, so ``cite_as`` is omitted rather than invented. 

959 """ 

960 if self.citation: 

961 return self.citation 

962 

963 names = [ 

964 c["name"] 

965 for c in (self.creators or []) 

966 if isinstance(c, dict) and c.get("name") 

967 ] 

968 author = ", ".join(names) if names else None 

969 

970 year = None 

971 for raw in (self.date_published, self.date_created): 

972 if not raw: 

973 continue 

974 try: 

975 year = datetime.fromisoformat(raw).year 

976 break 

977 except ValueError: 

978 continue 

979 

980 if author is None and year is None: 

981 return None 

982 

983 name = self.name or self.dataset_path.name 

984 author_part = author if author else "Unknown" 

985 year_part = f" ({year})." if year is not None else "" 

986 return f"{author_part}.{year_part} {name} [Data set]."

1007 """Generate and save Croissant metadata to a file. 

1008 

1009 Args: 

1010 output_path: Path where the JSON-LD metadata file will be written. 

1011 validate: If True (default), validates with mlcroissant before saving. 

1012 

1013 Raises:

1015 """

1018 output_file.parent.mkdir(parents=True, exist_ok=True) 

1019

1021 with tempfile.NamedTemporaryFile( 

1022 mode="w", suffix=".jsonld", delete=False 

1023 ) as tmp_file: 

1024 json.dump( 

1025 metadata_dict, 

1026 tmp_file, 

1027 indent=2, 

1028 ensure_ascii=False, 

1029 default=serialize_datetime,
