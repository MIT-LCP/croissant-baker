1"""FASTQ handler: the first read, and nothing the file says about anyone. 

2 

3A FASTQ is the same four lines repeated until the run is exhausted, and those 

4four lines carry no schema the rest of the file does not already repeat. The 

5one structural fact worth stating is how long a read is, and the first record 

6states it, so the first record is what is read and the file behind it is not. 

7 

8What the record also carries is a read name, and on an Illumina run that name 

9is instrument, run and flowcell identifiers spelled out. None of that is 

10structure, so none of it is emitted: not in the metadata, and not in the 

11description either. 

12 

13No RecordSet: sequencing reads are records of a run, not of a dataset schema. 

14What this handler produces is a described FileObject, through the 

15``description`` key the generator honours, exactly as the BAM handler does. 

16"""

18from typing import List

20from croissant_baker.handlers.base_handler import BuildResult, FileTypeHandler 

21from croissant_baker.handlers.utils import plural, read_prefix_chunks

24#: FASTQ has no IANA registration. The ``x-`` form follows ``text/x-vcf`` and 

25#: ``text/x-geo-soft``, already in the tree. 

26ENCODING_FORMAT = "text/x-fastq" 

27 

28#: The four lines of one record: name, sequence, separator, qualities. 

29RECORD_LINES = 4 

30 

31#: The character a record's name line opens with, and a separator line's own. 

32NAME_PREFIX = "@" 

33SEPARATOR_PREFIX = "+" 

34 

35#: Enough of the head to see the first record's shape. A read is tens to a few 

36#: hundred bases and a name is shorter still, so three lines of any real record 

37#: fit many times over; a peek that stops short of the third line simply leaves 

38#: the decision to the extension, and ``extract`` reports what it then finds. 

39CLAIM_BYTES = 4096 

40 

41#: The prefix the first record must end inside. Bounded rather than four lines 

42#: taken off the stream, which on a file holding no line ending at all reads 

43#: the whole of it, and reading the whole of it is the one thing this handler 

44#: exists not to do. A megabyte because a record is the read written twice, 

45#: once as bases and once as quality scores, and the longest reads anyone 

46#: produces are the hundreds of kilobases of a long-read platform; short-read 

47#: records, which are what most files hold, fit thousands of times over. 

48HEAD_BYTES = 1024 * 1024

51class FASTQHandler(FileTypeHandler): 

52 """Handler for FASTQ sequencing read files (``.fastq``, ``.fq``). 

53 

54 The first record is read and the rest of the file is not, so a run of a 

55 hundred million reads costs the same read as a run of one. 

56 

57 No RecordSet is emitted. The output is the FileObject the generator builds, 

58 carrying the description this handler wrote. 

59 """ 

60 

61 EXTENSIONS = (".fastq", ".fq") 

62 FORMAT_NAME = "FASTQ" 

63 ENCODING_FORMAT = ENCODING_FORMAT 

64 FORMAT_DESCRIPTION = "Read length of the first record; no read name or record count" 

65 

66 def claims(self, source: FileSource) -> bool: 

67 """Claim a file named like a FASTQ whose head is shaped like one. 

68 

69 Both conditions, because neither survives on its own. ``@`` is how a 

70 record's name line opens, and also how every line of a SAM header 

71 opens, so the first byte cannot decide alone; the extension cannot 

72 either, because it would claim any text a user happened to name 

73 ``.fq``. Together they are the record's own shape: a name line, then a 

74 sequence, then a separator on the third line. 

75 

76 A file that cannot be read peeks as ``b""`` and is therefore not 

77 claimed; that is 

78 :meth:`~croissant_baker.sources.FileSource.peek`'s contract, and no 

79 handler repeats it. 

80 """ 

81 if source.suffix not in self.EXTENSIONS: 

82 return False 

83 head = source.peek(CLAIM_BYTES) 

84 if not head.startswith(NAME_PREFIX.encode()): 

85 return False 

86 lines = head.splitlines() 

87 if len(lines) < 3: 87 ↛ 88line 87 didn't jump to line 88 because the condition on line 87 was never true

88 return True 

89 return lines[2].startswith(SEPARATOR_PREFIX.encode())

91 def extract(self, source: FileSource, **kwargs) -> dict: 

92 """Read the first record of a FASTQ, and stop there. 

93 

94 Args: 

95 source: The file, already decompressed. 

96 """ 

97 if not source.exists: 

98 raise FileNotFoundError( 

99 f"{self.FORMAT_NAME} file not found: {source.relative_path}" 

100 ) 

101 

102 name = str(source.relative_path) 

103 lines = self._read_first_record(source, name) 

104 read_length = self._first_read_length(lines, name) 

105 

106 return { 

107 "file_name": source.name, 

108 "file_size": source.size, 

109 "sha256": source.sha256, 

110 "encoding_format": self.ENCODING_FORMAT, 

111 "first_read_length": read_length, 

112 # The one thing this handler emits. Built here rather than in 

113 # build_croissant, which runs after the FileObject is staged, and 

114 # from the logical name, which is the only one extraction is given. 

115 "description": ( 

116 f"{self.FORMAT_NAME} sequencing reads {source.name} " 

117 f"(first read {plural(read_length, 'base')}). Described from " 

118 "its first record; no other record was read, and read names " 

119 "are not reported." 

120 ), 

121 }

123 def _read_first_record(self, source: FileSource, name: str) -> List[str]: 

124 """The first four lines, decoded and stripped of their endings. 

125 

126 Taken from a bounded prefix rather than as four lines off the stream: a 

127 stream iterated by line hands back the whole file as one line when the 

128 file holds no line ending, so a record that does not end inside the 

129 prefix is reported rather than read for. 

130 

131 Permissively decoded: a stray byte in a read name is not a reason to 

132 refuse a file whose structure is otherwise readable, and the name is 

133 the one part of the record nothing is emitted from anyway. 

134 """ 

135 head = b"" 

136 try: 

137 with source.open() as stream: 

138 for chunk in read_prefix_chunks(stream, HEAD_BYTES): 

139 head += chunk 

140 if head.count(b"\n") >= RECORD_LINES: 

141 break 

142 except UNREADABLE as exc: 

143 raise ValueError( 

144 f"Failed to read {self.FORMAT_NAME} file {name}: {exc}" 

145 ) from exc 

146 

147 if len(head) == HEAD_BYTES and head.count(b"\n") < RECORD_LINES: 

148 raise ValueError( 

149 f"Malformed {self.FORMAT_NAME} record in {name}: the first " 

150 f"record does not end within {HEAD_BYTES} bytes, so it is not " 

151 "the four lines this handler reads" 

152 ) 

153 lines = head.split(b"\n") 

154 # The tail after the last line ending is not a line. 

155 if lines and not lines[-1]: 

156 lines.pop() 

157 return [ 

158 line.decode("utf-8", "replace").rstrip("\r") 

159 for line in lines[:RECORD_LINES] 

160 ] 

161 

162 def _first_read_length(self, lines: List[str], name: str) -> int: 

163 """The length of the first read, or a refusal saying what is wrong. 

164 

165 Every refusal here is a record this handler cannot describe truthfully, 

166 and a wrong read length is worse than none: the file is reported with 

167 the reason instead. 

168 """ 

169 if not lines: 

170 raise ValueError( 

171 f"Empty {self.FORMAT_NAME} file: {name} holds no record to describe" 

172 ) 

173 if not lines[0].startswith(NAME_PREFIX): 

174 raise ValueError( 

175 f"Not a {self.FORMAT_NAME} file: the first line of {name} does " 

176 f"not open with '{NAME_PREFIX}'" 

177 ) 

178 if len(lines) < 3 or not lines[2].startswith(SEPARATOR_PREFIX): 

179 raise ValueError( 

180 f"Malformed {self.FORMAT_NAME} record in {name}: the third line " 

181 f"does not open with '{SEPARATOR_PREFIX}', so the first record " 

182 "does not fit the four lines this handler reads" 

183 ) 

184 sequence = lines[1] 

185 if len(lines) < RECORD_LINES: 

186 raise ValueError( 

187 f"Malformed {self.FORMAT_NAME} record in {name}: the first read " 

188 "ends before its quality line, so it is not a record to describe" 

189 ) 

190 if len(lines[3]) != len(sequence): 

191 raise ValueError( 

192 f"Malformed {self.FORMAT_NAME} record in {name}: the first read " 

193 f"is {len(sequence)} bases and carries {len(lines[3])} quality " 

194 "scores" 

195 ) 

196 return len(sequence)

198 def build_croissant(self, file_metas: list, file_ids: list) -> tuple: 

199 """Nothing: a FASTQ is described as a file, by the description it carries. 

200 

201 Sequencing reads are records of a run, not of a dataset schema, and a 

202 RecordSet naming columns no consumer can read through Croissant would 

203 be a promise nobody can keep. 

204 """ 

205 return BuildResult([], [])
