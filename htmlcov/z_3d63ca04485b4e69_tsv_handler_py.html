3from croissant_baker.handlers.csv_handler import CSVHandler

6class TSVHandler(CSVHandler): 

7 """ 

8 Handler for TSV files. 

9 

10 TSV is structurally identical to CSV with a tab delimiter. This handler 

11 inherits all type inference, row counting, and Croissant 

12 generation logic from CSVHandler. The only differences are the file 

13 extensions it claims and the delimiter passed to PyArrow. 

14 

15 Supported: .tsv, wrapped or not. Compression is resolved before this 

16 handler is asked. 

17 

18 To add another delimiter-separated format (e.g. pipe-separated): subclass 

19 CSVHandler, override _suffix() and _delimiter(), register the instance 

20 in registry.py. No other files need to change. 

21 """ 

22 

23 EXTENSIONS = (".tsv",) 

24 FORMAT_NAME = "TSV" 

25 FORMAT_DESCRIPTION = "Column names, inferred types, optional row count" 

26 

27 @staticmethod 

28 def _suffix() -> str: 

29 return ".tsv" 

30 

31 @staticmethod 

32 def _delimiter() -> str: 

33 return "\t" 

34 

35 @staticmethod 

36 def _encoding_format() -> str:
