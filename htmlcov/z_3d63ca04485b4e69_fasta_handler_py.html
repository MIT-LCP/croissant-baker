1"""FASTA handler: that the file is a FASTA, and nothing about what it holds. 

2 

3A FASTA is a description line and then bases, repeated. The description line is 

4the only thing in the file that says what format it is in, so it is read; its 

5content is a name a depositor chose, and for a per-sample assembly that name is 

6the sample, so the name is not emitted. The bases below it are never touched: 

7a reference genome is gigabytes of them and none of them is metadata. 

8 

9That leaves format and encoding, which is what this handler reports. No 

10RecordSet: bases are not records of a dataset schema. What it produces is a 

11described FileObject, through the ``description`` key the generator honours. 

12""" 

13 

14from croissant_baker.handlers.base_handler import BuildResult, FileTypeHandler 

15from croissant_baker.sources import UNREADABLE, FileSource 

16 

17#: The character a record description line opens with, and the whole of what 

18#: distinguishes a FASTA from any other text file. 

19RECORD_MARKER = ">" 

20 

21#: FASTA has no IANA registration. The ``x-`` form follows ``text/x-vcf``, 

22#: already in the tree. 

23ENCODING_FORMAT = "text/x-fasta" 

24 

25#: How much of the head is pulled to reach the end of the first line. Bounded 

26#: rather than a plain ``readline``, which on a file holding no newline at all 

27#: reads the whole of it, and reading the whole of it is the one thing this 

28#: handler exists not to do. Nothing past the first newline is looked at, and a 

29#: description line longer than this is still recognised by its opening bytes, 

30#: which is all that is asked of it. 

31HEAD_BYTES = 4096 

32 

33 

34class FASTAHandler(FileTypeHandler): 

35 """Handler for FASTA sequence files (``.fa``, ``.fasta``, ``.fna``). 

36 

37 One line is read and then discarded. The output is the FileObject the 

38 generator builds, carrying the description this handler wrote: the file is 

39 a FASTA, and that is the claim it makes. 

40 """ 

41 

42 EXTENSIONS = (".fa", ".fasta", ".fna") 

43 FORMAT_NAME = "FASTA" 

44 FORMAT_DESCRIPTION = "Format and encoding; record names and sequences are not read" 

45 

46 def claims(self, source: FileSource) -> bool: 

47 """Claim a declared FASTA extension whose first byte opens a record. 

48 

49 Both halves are needed, and neither would do on its own. ``>`` is a 

50 single character that a quoted email, a shell transcript and a diff all 

51 begin with, which is too little to own a file on. The extension alone 

52 would claim any text a user happened to name ``.fa``, and the format 

53 has no other marker to fall back on: past that first byte a FASTA is 

54 letters, which is what an unrelated text file is too. 

55 

56 A file that cannot be read peeks as ``b""`` and is therefore not 

57 claimed, corrupt wrappers included; that is 

58 :meth:`~croissant_baker.sources.FileSource.peek`'s contract, and no 

59 handler repeats it. 

60 """ 

61 if source.suffix not in self.EXTENSIONS: 

62 return False 

63 return source.peek(len(RECORD_MARKER)) == RECORD_MARKER.encode() 

64 

65 def extract(self, source: FileSource, **kwargs) -> dict: 

66 """Read the first record's description line, and stop there. 

67 

68 The line is read to confirm the format declares a record and is then 

69 forgotten: nothing it carries reaches the metadata. What is emitted is 

70 the file's own identity, which the bytes on disk state. 

71 """ 

72 if not source.exists: 

73 raise FileNotFoundError( 

74 f"{self.FORMAT_NAME} file not found: {source.relative_path}" 

75 ) 

76 

77 self._read_first_line(source) 

78 

79 return { 

80 "file_name": source.name, 

81 "file_size": source.size, 

82 "sha256": source.sha256, 

83 "encoding_format": ENCODING_FORMAT, 

84 # The one thing this handler emits, built from the logical name, 

85 # which is the only name extraction is given. 

86 "description": ( 

87 f"{self.FORMAT_NAME} sequence file {source.name}. Described " 

88 "from its first record line; record names and sequences are " 

89 "not read." 

90 ), 

91 } 

92 

93 def _read_first_line(self, source: FileSource) -> None: 

94 """Refuse a head that does not open a record, naming the file and why. 

95 

96 Decoded permissively: a description line is ASCII by convention and 

97 nothing here is emitted, so a stray byte in a comment is not a reason 

98 to refuse a file whose format is otherwise stated. 

99 """ 

100 name = source.relative_path 

101 try: 

102 with source.open() as stream: 

103 head = stream.read(HEAD_BYTES) 

104 except UNREADABLE as exc: 

105 raise ValueError( 

106 f"Failed to read {self.FORMAT_NAME} file {name}: {exc}" 

107 ) from exc 

108 

109 if not head: 

110 raise ValueError( 

111 f"Not a {self.FORMAT_NAME} file: {name} is empty, so it " 

112 "declares no record" 

113 ) 

114 line = head.decode("utf-8", "replace").split("\n", 1)[0].rstrip("\r") 

115 if not line.startswith(RECORD_MARKER): 

116 raise ValueError( 

117 f"Not a {self.FORMAT_NAME} file: {name} does not open with the " 

118 f"'{RECORD_MARKER}' that starts a record description line" 

119 ) 

120 if not line[len(RECORD_MARKER) :].strip(): 

121 raise ValueError( 

122 f"Incomplete {self.FORMAT_NAME} header in {name}: the first " 

123 f"line is a bare '{RECORD_MARKER}', which names no record" 

124 ) 

125 

126 def build_croissant(self, file_metas: list, file_ids: list) -> tuple: 

127 """Nothing: a FASTA is described as a file, by the description it carries. 

128 

129 Bases are records of a genome, not of a dataset schema, and a RecordSet 

130 naming fields no consumer can read through Croissant would be a promise 

131 nobody can keep. The same reasoning as BAM. 

132 """ 

133 return BuildResult([], [])
