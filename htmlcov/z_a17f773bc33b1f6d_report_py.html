36@dataclass(frozen=True) 

37class ScanReport: 

38 """A view over resolved scan entries: what was described, and what was not. 

39 

40 :meth:`summary_lines` is for the terminal, and its length depends on how 

41 many *kinds* of problem occurred, never on how many files did. A 

42 diagnostic is counted there the same way: one line per code. 

43 :meth:`to_dict` carries the per-file detail, for ``--report`` and for 

44 downstream tools checking coverage. 

45 """ 

46 

47 entries: List[ScanEntry] = field(default_factory=list) 

48 

49 @property 

50 def total(self) -> int: 

51 """How many files the scan found.""" 

52 return len(self.entries) 

53 

54 @property 

55 def described(self) -> List[ScanEntry]: 

56 """Entries a handler described and whose nodes were assembled.""" 

57 return [e for e in self.entries if e.outcome is Outcome.DESCRIBED] 

58 

59 @property 

60 def linked(self) -> List[ScanEntry]: 

61 """Entries carried as another described file in a different form.""" 

62 return [e for e in self.entries if e.outcome is Outcome.LINKED] 

63 

64 @property 

65 def referenced(self) -> List[ScanEntry]: 

66 """Entries another file's handler put in the document.""" 

67 return [e for e in self.entries if e.outcome is Outcome.REFERENCED]

69 @property 

70 def undescribed(self) -> List[ScanEntry]: 

71 """Entries the document does not carry, in scan order. 

72 

73 Membership of the document, not of a record set: a file with a 

74 FileObject is in there whether it got there on its own 

75 (``DESCRIBED``), as another form of a described file (``LINKED``), or 

76 as part of a multi-file record (``REFERENCED``). 

77 """ 

78 return [e for e in self.entries if e.outcome not in _IN_DOCUMENT] 

79 

80 @property 

81 def diagnosed(self) -> List[ScanEntry]: 

82 """Entries carrying a diagnostic, in scan order, whatever the outcome.""" 

83 return [e for e in self.entries if e.diagnostics]

85 def diagnostic_counts(self) -> Dict[DiagnosticCode, int]: 

86 """Number of diagnostics per code, in declaration order. 

87 

88 Per diagnostic, not per file: one workbook can leave four sheets out. 

89 """ 

90 tally = Counter(d.code for e in self.entries for d in e.diagnostics) 

91 return {c: tally[c] for c in DiagnosticCode if tally[c]} 

92 

93 def counts(self) -> Dict[Reason, int]: 

94 """Number of entries per reason, in declaration order. 

95 

96 Over the undescribed only: a reason answers why a file is not in the 

97 document, so counting one against a file that is in there says the 

98 opposite of what it means. A linked file keeps its own reason — that 

99 is the evidence it was linked on — and is not counted here. 

100 """ 

101 tally = Counter(e.reason for e in self.undescribed if e.reason is not None) 

102 return {r: tally[r] for r in Reason if tally[r]}

135

138 are the three ways into the document and sum with ``undescribed`` to 

139 ``total``; ``by_reason`` accounts for the ``undescribed`` alone. 

140 

141 ``by_diagnostic`` is orthogonal to all of them, and a file's 

142 ``diagnostics`` key appears only when it has one. 

143 """ 

144 return {
