1"""A streaming reader for the GEO SOFT line grammar. 

2 

3Standard library only, and nothing from the rest of the tree. The grammar is at 

4https://www.ncbi.nlm.nih.gov/geo/info/soft.html; what the baker does with the 

5result is in ``docs/user-guide/supported-formats.md``. 

6 

7One fact about the format sets the shape of the code: **the tables are the 

8file.** 99.96 % of a 50 MiB family export lies between ``!*_table_begin`` and 

9``!*_table_end``, most of it in a single platform annotation table at ~2 KiB a 

10row. Rows are sampled up to a bound per column signature and then skipped, so 

11peak memory is set by the sample rather than by the deposit. 

12""" 

13 

14from __future__ import annotations 

15 

16import re 

17from dataclasses import dataclass, field 

18from typing import Iterable, Optional 

19 

20#: Rows sampled per distinct table signature, for the caller's type inference. 

21SAMPLE_ROWS = 500 

22 

23#: A ceiling on that same sample, because a row is not a bounded thing: GPL96 

24#: runs ~2 KiB a row, so 500 rows is already 1 MiB. The header is kept whatever 

25#: its size, which is the one way the buffer exceeds this. 

26SAMPLE_BYTES = 1 << 20 

27 

28# Series tables may have a title after '='. Tabs still distinguish table cells 

29# from control lines, including cells whose text resembles an end marker. 

30_TABLE_MARKER = re.compile( 

31 r"!([a-z]+)_table_(begin|end)(?:[ ]*=[ ]*([^\t]*))?", re.IGNORECASE 

32) 

33_ENTITY = re.compile(r"([A-Za-z][A-Za-z0-9_]*)\s*=\s*(\S+)\s*") 

34 

35#: ``!Sample_characteristics_ch1`` once its ``Sample_`` prefix has come off. 

36_CHARACTERISTICS = re.compile(r"^characteristics(?:_ch(\w+))?$", re.IGNORECASE) 

37 

38#: ``!Sample_data_row_count = 22283``, likewise prefix-stripped. Every sample of 

39#: every real export tested declares one, so no table body is ever counted. 

40_ROW_COUNT = "data_row_count" 

41 

42_KEY_SEPARATOR = ":" 

43 

44#: What a partial parse can mean. An attribute block has no closing marker, so a 

45#: file ending after a complete attribute line is *not* partial. 

46TABLE_UNCLOSED = "a data table was still open at end of file" 

47UNDECODABLE = "{count} line(s) could not be decoded"

53 

54 ``names`` maps each name to whether it repeated *inside a single entity* —

68@dataclass 

69class Table: 

70 """One column signature, and every entity that declared it. 

71 

72 ``rows`` is summed from the ``!*_data_row_count`` declarations and is 

73 trustworthy only when ``rows_declared`` reaches ``entities``. 

74 """ 

75 

76 kind: str 

77 columns: tuple 

78 title: str = "" 

79 #: Column name -> its ``#COLUMN`` line, verbatim. First non-empty one wins. 

80 column_lines: dict = field(default_factory=dict) 

81 entities: int = 0 

82 rows: int = 0 

83 rows_declared: int = 0 

84 #: Header plus a bounded row sample, as UTF-8 TSV, for the caller to type. 

85 #: Filled across every entity of this signature until the bound is spent, 

86 #: because a deposit may declare an empty table before a populated one. A 

87 #: ``bytearray`` since ``bytes += bytes`` copies the whole buffer per row. 

88 sample: bytearray = field(default_factory=bytearray, repr=False) 

89 _sampled_rows: int = field(default=0, repr=False)

91 @property 

92 def rows_known(self) -> bool: 

93 """Whether every contributing entity declared its row count."""

100 

101 ``kinds`` holds only the entity kinds present: the parser reports every kind

107 #: Characteristics lines that were not ``key: value``. The convention is a 

108 #: convention, not a grammar.

119def parse( 

120 lines: Iterable[bytes], 

121 *, 

122 sample_rows: int = SAMPLE_ROWS, 

123 sample_bytes: int = SAMPLE_BYTES, 

124) -> SoftFile: 

125 """Read one SOFT export in a single forward pass. 

126 

127 Args: 

128 lines: The decompressed file, iterated as raw byte lines. 

129 sample_rows: Table rows to buffer per distinct column signature. 

130 sample_bytes: A ceiling on that same buffer. 

131 

132 Returns: 

133 The deposit's shape. An input with no ``^`` entity line yields no kinds, 

134 which distinguishes non-SOFT input from an export with little metadata. 

135 

136 Raises: 

137 ValueError: An entity declaration or table marker is malformed. 

138 """ 

139 state = _State(sample_rows, sample_bytes) 

140 for raw in lines: 

141 state.feed(raw) 

142 return state.finish()

165 self._entity_counts: dict = {}

182 """Consume one raw line.""" 

183 line = self._decode(raw) 

184 if self._first_line: 

185 line = line.removeprefix("\ufeff") 

186 self._first_line = False

194 self._close_table()

196 self._start_table(line) 

197 else: 

198 self._sample_row(line) 

199 return

222 """Report what the pass found, and whether it reached the end.""" 

223 incomplete = [] 

224 if self._in_table: 

225 incomplete.append(TABLE_UNCLOSED) 

226 if self._undecodable: 

227 incomplete.append(UNDECODABLE.format(count=self._undecodable))

236 def _decode(self, raw: bytes) -> str: 

237 """One line as text, having counted it if a byte would not decode. 

238 

239 GEO declares no encoding. Every real export tested is valid UTF-8 — the 

240 ``37ºC`` in a 2004 deposit is ``0xC2 0xBA``, and latin-1 is what 

241 corrupts it to ``37ÂºC`` — but the format guarantees nothing. 

242 """ 

243 try: 

244 return raw.decode("utf-8").rstrip("\r\n") 

245 except UnicodeDecodeError: 

246 self._undecodable += 1 

247 return raw.decode("utf-8", "replace").rstrip("\r\n")

249 def _open_entity(self, rest: str) -> None: 

250 match = _ENTITY.fullmatch(rest) 

251 if match is None: 

252 raise ValueError("Malformed GEO SOFT '^ENTITY = ACCESSION' line") 

253 kind, accession = match.groups() 

254 kind = kind.upper() 

255 entity = (kind, accession) 

256 group = self._out.kinds.setdefault(kind, FieldGroup()) 

257 if entity not in self._entity_counts: 

258 group.entities += 1 

259 self._kind = kind 

260 self._prefix = f"{kind.lower()}_" 

261 self._counts = self._entity_counts.setdefault(entity, {}) 

262 self._char_counts = self._entity_char_counts.setdefault(entity, {}) 

263 self._fallback_counts = self._entity_fallback_counts.setdefault(entity, {}) 

264 self._column_lines = {} 

265 self._declared_rows = None

267 def _attribute(self, rest: str) -> None: 

268 if self._kind is None: 

269 return # a SOFT file opens with ^DATABASE before any ! 

270 

271 raw_name, _, value = rest.partition("=") 

272 name = self._strip_prefix(raw_name.strip()) 

273 if not name: 

274 return # `! = value` names nothing, and a field needs a name 

275 

276 if name.lower() == _ROW_COUNT: 

277 # Still an attribute, and so still a field: a name the deposit uses 

278 # does not stop being one because the parser reads its value. 

279 self._declared_rows = _as_int(value.strip()) 

280 

281 channel = _characteristic_channel(name) 

282 if channel is not None and self._kind == "SAMPLE": 

283 self._characteristic(name, channel, value.lstrip()) 

284 return 

285 

286 count = self._counts[name] = self._counts.get(name, 0) + 1 

287 self._out.kinds[self._kind].observe(name, count > 1)

289 def _characteristic(self, attribute: str, channel: str, value: str) -> None: 

290 # An empty value, as in ``diagnosis:``, still declares its key. 

291 key, sep, _ = value.partition(_KEY_SEPARATOR) 

292 if not sep or not key.strip(): 

293 self._out.unparsed += 1 

294 count = self._fallback_counts[attribute] = ( 

295 self._fallback_counts.get(attribute, 0) + 1 

296 ) 

297 was = self._out.fallbacks.get(attribute, False)

299 return

302 count = self._char_counts[slot] = self._char_counts.get(slot, 0) + 1

316 def _start_table(self, header: str) -> None: 

317 columns = tuple(header.split("\t")) 

318 signature = (self._kind, self._table_title, columns) 

319 

320 table = self._by_signature.get(signature) 

321 if table is None: 

322 table = Table( 

323 kind=self._kind or "", columns=columns, title=self._table_title 

324 ) 

325 table.sample += header.encode("utf-8") + b"\n" 

326 self._by_signature[signature] = table 

327 self._out.tables.append(table) 

328 

329 for column, line in self._column_lines.items(): 

330 table.column_lines.setdefault(column, line) 

331 table.entities += 1 

332 if self._declared_rows is not None: 

333 table.rows += self._declared_rows 

334 table.rows_declared += 1 

335 # A later table under this entity needs its own declaration rather 

336 # than reusing the preceding table's row count. 

337 self._declared_rows = None 

338 self._table = table 

339 

340 def _sample_row(self, line: str) -> None: 

341 table = self._table 

342 if table is None or table._sampled_rows >= self._sample_rows: 

343 return 

344 # GEO's GSE2034 series matrix rows have an extra trailing tab. Ignore 

345 # only empty cells beyond the declared header, never populated extras 

346 # or empty cells belonging to declared columns. 

347 cells = line.split("\t") 

348 width = len(table.columns) 

349 if len(cells) > width and not any(cells[width:]): 

350 line = "\t".join(cells[:width]) 

351 row = line.encode("utf-8") + b"\n" 

352 if len(table.sample) + len(row) > self._sample_bytes: 

353 # Too big for what is left of the budget. Skipped rather than 

354 # final: a smaller row later — or from another entity of this 

355 # signature — is still evidence, and a signature left with only 

356 # its header types every numeric column as text. 

357 return 

358 table.sample += row 

359 table._sampled_rows += 1

368 def _strip_prefix(self, raw: str) -> str: 

369 """``Sample_title`` under a ``^SAMPLE`` becomes ``title``. 

370 

371 Only the open entity's own prefix comes off, so a stray ``!Sample_foo`` 

372 inside a ``^PLATFORM`` is not renamed to something never written. 

373 """ 

374 if self._prefix and raw.lower().startswith(self._prefix): 

375 return raw[len(self._prefix) :] or raw 

376 return raw

378 def _name_characteristics(self) -> None: 

379 """Turn the collected ``(channel, key)`` slots into field names. 

380 

381 Channel 1 (or an omitted channel) keeps the bare key, regardless of 

382 other samples' channels. Later channels always carry their prefix. 

383 Disambiguate collisions so a submitter's ``ch2_tissue`` and channel 2's 

384 ``tissue`` both survive. Suffixes never displace another literal key. 

385 """ 

386 group = self._out.characteristics 

387 group.entities = self._out.kinds.get("SAMPLE", FieldGroup()).entities 

388 names = [ 

389 ( 

390 key if channel == "1" else f"ch{channel}_{key}", 

391 repeated, 

392 f"'{key}' in !Sample_characteristics_ch{channel}", 

393 0 if channel == "1" else 1, 

394 ) 

395 for (channel, key), repeated in self._characteristics.items() 

396 ] + [ 

397 (attribute, repeated, f"unparsed !Sample_{attribute}", 2) 

398 for attribute, repeated in self._out.fallbacks.items() 

399 ] 

400 reserved = {name for name, _, _, _ in names} 

401 used: set = set() 

402 assigned: dict = {} 

403 # A literal channel-1 key keeps its spelling even when another sample 

404 # declared the colliding channel-2 name first. 

405 for i in sorted(range(len(names)), key=lambda i: names[i][3]): 

406 name = names[i][0] 

407 candidate = name 

408 if candidate in used: 

409 n = 2 

410 while f"{name}__{n}" in reserved or f"{name}__{n}" in used: 

411 n += 1 

412 candidate = f"{name}__{n}" 

413 used.add(candidate) 

414 assigned[i] = candidate 

415 for i, (name, repeated, origin, _) in enumerate(names): 

416 candidate = assigned[i] 

417 group.observe(candidate, repeated) 

418 if candidate != name: 

419 group.origins[candidate] = origin

422def _table_marker(line: str) -> Optional[tuple]: 

423 """The entity kind, action and optional title of a whole marker line. 

424 

425 Spaces are forgiven, tabs are not: the line ending is already off, so what 

426 remains to strip is either padding around a marker or a TSV delimiter, and 

427 ``!sample_table_end\t`` is a two-column row whose second cell is empty. 

428 """ 

429 match = _TABLE_MARKER.fullmatch(line.strip(" ")) 

430 if match is None: 

431 return None 

432 kind, action, title = match.groups() 

433 return kind.upper(), action.lower(), (title or "").strip()
