1"""A local stdio MCP server: a second front door to the same pipeline. 

2 

3The three tools below are plain functions with typed signatures, independent of 

4any transport, so they can be called directly by tests and by other Python 

5code. :func:`build_server` is the only place that knows about the MCP SDK, and 

6it is imported lazily so the package stays installable without the optional 

7``mcp`` extra. 

8 

9The surface is deliberately narrow. ``bake`` accepts the semantic fields a 

10human would type and nothing else: an agent can supply only what a person 

11could, and the structural layer is untouched. The ``ScanReport`` counters come 

12back with every refusal counted by reason, and the full per-file report is 

13written next to the output. There is no fetch, search or upload tool, and no 

14HTTP transport, so the local-first invariant holds. 

15 

16Alongside the tools the server publishes two read-only resources: the packaged 

17Agent Skill and the RAI config template it points at. A client that has the 

18tools but not the skill would otherwise have to be told separately how to use 

19them, and could not open a file the skill names. 

20""" 

21

51def _skill_file(*parts: str) -> str: 

52 """Return the text of a file in the bundled skill directory. 

53

75def dry_run( 

76 input_dir: str, 

77 include: Optional[List[str]] = None, 

78 exclude: Optional[List[str]] = None, 

79 output: Optional[str] = None, 

80) -> dict: 

81 """Report what a bake of ``input_dir`` would describe, without baking it. 

82 

83 The counters are the dry run's own, not :meth:`ScanReport.to_dict`'s. That 

84 summary is written for a completed bake, where a file is either in the 

85 document or accounted for by a reason; a dry run describes nothing, so every 

86 file would land in ``undescribed`` and a directory that bakes cleanly would 

87 report as describing none of it. The counts here mirror what the CLI's 

88 ``--dry-run`` prints: claimed and unclaimed. 

89 

90 Args: 

91 input_dir: Directory containing the dataset files. 

92 include: Optional glob patterns; only matching files are scanned. 

93 exclude: Optional glob patterns; matching files are skipped. 

94 output: The output path the bake will be given. When it falls inside 

95 ``input_dir``, that file and its report are skipped exactly as 

96 bake skips them, so a dry run after an earlier bake counts what 

97 the next bake will. 

98 

99 Returns: 

100 ``total``, ``would_process`` and ``unclaimed`` counts, ``by_reason`` 

101 accounting for the unclaimed alone, and ``files``: each unclaimed 

102 file with its reason and a human-readable detail. Claimed files are 

103 counted and not listed, so a large tree does not flood the caller 

104 with entries that need no decision. 

105 """ 

106 skip = _own_files_under(input_dir, *_bake_files(output)) if output else None 

107 entries = dry_run_entries(input_dir, include, exclude, skip) 

108 claimed = [e for e in entries if e.outcome is Outcome.WOULD_PROCESS] 

109 unclaimed = [e for e in entries if e.outcome is Outcome.UNCLAIMED] 

110 tally = Counter(e.reason for e in unclaimed if e.reason is not None) 

111 return { 

112 "total": len(entries), 

113 "would_process": len(claimed), 

114 "unclaimed": len(unclaimed), 

115 "by_reason": {r.value: tally[r] for r in Reason if tally[r]}, 

116 "files": ScanReport(unclaimed).to_dict()["files"], 

117 }

120def bake( 

121 input_dir: str, 

122 output: str, 

123 name: str, 

124 description: str, 

125 license: str, 

126 creators: List[str], 

127 url: Optional[str] = None, 

128 citation: Optional[str] = None, 

129 date_published: Optional[str] = None, 

130 detect_references: bool = False, 

131 include: Optional[List[str]] = None, 

132 exclude: Optional[List[str]] = None, 

133) -> dict: 

134 """Generate Croissant metadata for ``input_dir`` and write it to ``output``. 

135 

136 Args: 

137 input_dir: Directory containing the dataset files. 

138 output: Path of the ``.jsonld`` file to write. 

139 name: Dataset name. 

140 description: Dataset description. 

141 license: Dataset license, preferably as a URL. 

142 creators: Creators, each ``"Name"``, ``"Name,email"`` or 

143 ``"Name,email,url"``, the same format the CLI's ``--creator`` 

144 accepts. 

145 url: Optional dataset homepage. 

146 citation: Optional citation text. 

147 date_published: Optional publication date in ISO format, such as 

148 ``2024-01-02``. The spec expects one, and the CLI's 

149 ``--date-published`` is the same field. 

150 detect_references: Detect foreign keys between record sets. 

151 include: Optional glob patterns; only matching files are described. 

152 exclude: Optional glob patterns; matching files are skipped. 

153 

154 Returns: 

155 ``output`` and ``report_path``, both absolute, then the completed 

156 bake's counters from :meth:`ScanReport.to_dict`: ``total``, 

157 ``described``, ``linked``, ``referenced``, ``undescribed``, 

158 ``by_reason`` and ``by_diagnostic``. The per-file list is left out, 

159 since a large tree would flood the caller with one entry per file, as 

160 the CLI's bounded summary avoids too. It is written instead to 

161 ``report_path``, next to the output: ``out.jsonld`` gets 

162 ``out.report.json``. A relative ``output`` resolves against the 

163 server's working directory, which the caller may not know, so the 

164 paths returned are the ones really written. When either file falls 

165 inside ``input_dir`` it is left out of the scan, so the next bake 

166 does not describe it. 

167 

168 Raises: 

169 ValueError: If there is no creator, a creator has a blank name part, 

170 ``date_published`` is not an ISO date, or the document fails 

171 ``mlcroissant`` validation; in each case nothing is written. 

172 """ 

173 check_iso_dates("date_published", [date_published] if date_published else None) 

174 output_path, report_path = _bake_files(output) 

175 

176 generator = MetadataGenerator( 

177 dataset_path=input_dir, 

178 name=name, 

179 description=description, 

180 url=url, 

181 license=license, 

182 citation=citation, 

183 date_published=date_published, 

184 creators=parse_creators(creators, cli=False), 

185 detect_references=detect_references, 

186 includes=include, 

187 excludes=exclude, 

188 skip_paths=_own_files_under(input_dir, output_path, report_path), 

189 ) 

190 metadata_dict = generator.generate_metadata() 

191 save_dict(metadata_dict, str(output_path), validate=True) 

192 write_scan_report(generator.scan_report, report_path) 

193 summary = generator.scan_report.to_dict() 

194 del summary["files"] 

195 return {"output": str(output_path), "report_path": str(report_path), **summary}

198def _bake_files(output: str) -> Tuple[Path, Path]: 

199 """The output and the report a bake to ``output`` writes, resolved.""" 

200 output_path = Path(output).resolve() 

201 return output_path, output_path.with_suffix(".report.json")

204def _own_files_under(input_dir: str, *paths: Path) -> List[Path]: 

205 """The files a bake writes inside its own input, relative to it. 

206 

207 A bake written into the dataset directory would otherwise describe its 

208 own output and report on the next run, and the report, being JSON, is 

209 claimed and refused. Each one is skipped by its exact path, so data of 

210 the same name elsewhere in the tree is still scanned. 

211 """ 

212 root = Path(input_dir).resolve() 

213 return [path.relative_to(root) for path in paths if path.is_relative_to(root)]

216def validate(path: str) -> dict: 

217 """Check that a Croissant file constructs under ``mlcroissant``. 

218 

219 Only a file on this machine is read. ``mlcroissant`` would fetch a URL 

220 it is given, and a URL that reached this tool from untrusted text would 

221 then become a request from the user's machine. 

222 

223 Args: 

224 path: Path to a local Croissant JSON-LD file. 

225 

226 Returns: 

227 ``{"valid": True}``, or ``{"valid": False, "error": <message>}`` when 

228 ``mlcroissant`` read the file and refused it. 

229 

230 Raises: 

231 FileNotFoundError: If ``path`` is not a local file, including a URL. 

232 Kept apart from ``valid: False``, which is about a document that 

233 exists. 

234 OSError: If the file cannot be opened. 

235 """ 

236 import mlcroissant as mlc 

237 

238 local = Path(path).resolve() 

239 if not local.is_file(): 

240 raise FileNotFoundError( 

241 f"{path} is not a local file. validate reads files on this " 

242 "machine only, and never fetches a URL." 

243 ) 

244 with open(local, "rb"): 

245 pass 

246 

247 try: 

248 mlc.Dataset(str(local)) 

249 except Exception as exc: 

250 return {"valid": False, "error": str(exc)} 

251 return {"valid": True}

254def _reported_to_client(tool: Callable[..., Any], error: type) -> Callable[..., Any]: 

255 """Wrap ``tool`` so a refusal it raises reaches the client as ``error``. 

256 

257 The SDK treats any other exception from a tool as a crash and hides its 

258 text, so the client would learn only that the tool failed. A ``ValueError`` 

259 here is a refusal of the caller's input, such as a creator with no name, 

260 and an ``OSError`` is a path the caller named that cannot be read or 

261 written; either message says what to fix. 

262 """ 

263 

264 @functools.wraps(tool) 

265 def wrapper(*args: Any, **kwargs: Any) -> Any: 

266 try: 

267 return tool(*args, **kwargs) 

268 except (ValueError, OSError) as exc: 

269 raise error(str(exc)) from exc 

270 

271 return wrapper

274def build_server() -> Any: 

275 """Build the MCP server with the three tools and the two resources. 

276 

277 Returns: 

278 An ``MCPServer`` from the ``mcp`` package, imported here so the 

279 dependency stays optional. 

280 

281 Raises: 

282 ImportError: If the optional ``mcp`` extra is not installed. 

283 """ 

284 from mcp.server.mcpserver import MCPServer 

285 from mcp.server.mcpserver.exceptions import ToolError 

286 from mcp.types import ToolAnnotations 

287 

288 # The hints let a client approve the two tools that only read on its own 

289 # and ask before bake, which writes the output, the report and any 

290 # missing parent directories, replacing files already there. None of the 

291 # three reaches beyond the local machine. 

292 read_only = ToolAnnotations(read_only_hint=True, open_world_hint=False) 

293 writes = ToolAnnotations( 

294 read_only_hint=False, destructive_hint=True, open_world_hint=False 

295 ) 

296 

297 server = MCPServer(SERVER_NAME, version=get_version()) 

298 for tool, hints in ( 

299 (dry_run, read_only), 

300 (bake, writes), 

301 (validate, read_only), 

302 ): 

303 server.add_tool(_reported_to_client(tool, ToolError), annotations=hints) 

304 server.resource( 

305 SKILL_URI, 

306 name="croissant-baker-skill", 

307 title="Croissant Baker Agent Skill", 

308 description=( 

309 "How to drive croissant-baker: the dry-run-bake-validate loop, " 

310 "which fields are inferred and which must be asked for, and the " 

311 "gotchas." 

312 ), 

313 mime_type="text/markdown", 

314 )(skill_markdown) 

315 server.resource( 

316 RAI_TEMPLATE_URI, 

317 name="croissant-baker-rai-template", 

318 title="Croissant Baker RAI config template", 

319 description=( 

320 "Starting point for --rai-config: every key the loader accepts, " 

321 "with a comment on what belongs in each." 

322 ), 

323 mime_type="application/yaml", 

324 )(rai_template_yaml) 

325 return server

328def serve() -> None:
