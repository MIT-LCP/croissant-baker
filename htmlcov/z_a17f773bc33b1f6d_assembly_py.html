18def _encoding_formats(format_media_type: str, relative_path: str) -> List[str]: 

19 """The media types describing one file: its format, then its wrapper. 

20 

21 A gzipped CSV is ``["text/csv", "application/gzip"]``. Croissant 1.1 gives 

22 ``sc:encodingFormat`` cardinality MANY, and this shape is proposal 2 of 

23 mlcommons/croissant#635, which is still open on how single-file compression 

24 should be expressed. The alternatives are worse: ``+``-concatenation 

25 (proposal 1) produces a string no consumer matches, and ``containedIn`` is 

26 the spec's mechanism for archives, whose extraction mlcroissant gates on 

27 tar/zip. 

28 

29 Order matters. mlcroissant decides to decompress from the filename suffix, 

30 not from this list, and re-applies it on each iteration until a media type 

31 it recognises returns — so the format must come first. Reversed, a ``.gz`` 

32 file is gunzipped twice. 

33 

34 Handlers report only the format; the wrapper is the pipeline's to know. 

35 """ 

36 wrapper = compression.compression_for(Path(relative_path).name) 

37 if wrapper is None: 

38 return [format_media_type] 

39 return [format_media_type, wrapper.media_type]

46def _matches(pattern: str, path: str) -> bool: 

47 """Whether a dataset-relative ``path`` is covered by an ``includes`` pattern. 

48 

49 Directory-scoped, which is what a Croissant ``includes`` means and what 

50 ``Path.match`` does not do: ``Path("other/data/x.parquet").match( 

51 "data/*.parquet")`` is ``True``, because ``match`` anchors on the right. 

52 Here ``*`` and ``?`` stay inside one path segment, and only ``**`` crosses 

53 ``/`` — matching zero or more segments, so ``**/*.png`` covers ``a.png`` as 

54 well as ``deep/a.png``. 

55 

56 Case-insensitive, because handler dispatch is: ``FileSource.suffix`` 

57 lowercases, so ``pixel.PNG`` is described by the same handler that declares 

58 ``**/*.png`` and belongs to the FileSet that pattern stands for. 

59 """ 

60 return _match_segments(pattern.lower().split("/"), path.lower().split("/"))

63def _match_segments(pattern: List[str], parts: List[str]) -> bool: 

64 if not pattern: 

65 return not parts 

66 head, rest = pattern[0], pattern[1:] 

67 if head == "**": 

68 return any(_match_segments(rest, parts[i:]) for i in range(len(parts) + 1)) 

69 if not parts: 

70 return False 

71 return fnmatch.fnmatchcase(parts[0], head) and _match_segments(rest, parts[1:])

74def _wrappers_among(stored: Iterable[str]) -> List: 

75 """The compressions used by ``stored``, in registry order. 

76 

77 Registry order rather than set order keeps the output stable. 

78 """ 

79 present = { 

80 comp.suffix 

81 for name in stored 

82 if (comp := compression.compression_for(Path(name).name)) 

83 } 

84 return [c for c in compression.compressions() if c.suffix in present]

87def _resolve_file_sets(file_sets: list, stored_paths: dict, entries: list) -> list: 

88 """Point each FileSet at the files as stored, and at the wrappers they use. 

89 

90 A handler names its files logically; only the generator knows what is on 

91 disk. An exact name becomes the stored path or paths it stands for, and a 

92 pattern gains one variant per compression used by the files *that pattern* 

93 matches. Expanding an exact name as if it were a pattern would append a 

94 second suffix. An exact name is matched exactly, not case-insensitively as 

95 a pattern is: it comes from a handler's own metadata, so it already carries 

96 the spelling the file is stored under. 

97 

98 Each FileSet is resolved against its own members, never against the batch: 

99 one handler describes every file of its format in the dataset, so a wrapper 

100 anywhere would otherwise reach a FileSet whose own directory holds none — 

101 claiming ``application/gzip`` and an include matching no file. 

102 

103 Exclusions are resolved first, including duplicates of excluded files. 

104 They do not contribute to the FileSet's compression types or include 

105 variants, and no duplicate may reintroduce an excluded member. 

106 

107 Membership follows the files, not the glob text. An entry belongs to a 

108 FileSet when a pattern matches the name it is stored under, or when it 

109 duplicates an entry that belongs: a twin stored as ``pixel.jpeg.gz`` rides 

110 with the ``pixel.jpg`` it links to, and no glob the handler declares spells 

111 ``.jpeg``. 

112 

113 Args: 

114 file_sets: FileSets from one handler batch. Mutated in place. 

115 stored_paths: Logical dataset-relative path to the stored paths sharing 

116 it. A plain file and its wrapper share one logical key. 

117 entries: The scan entries this batch describes, including the linked 

118 duplicates riding along with them. 

119 

120 Returns: 

121 ``file_sets``, for chaining. 

122 """ 

123 by_logical: dict = {} 

124 for entry in entries: 

125 logical = str(entry.path.with_name(compression.logical_name(entry.path.name))) 

126 by_logical.setdefault(logical, []).append(entry) 

127 

128 for file_set in file_sets: 

129 includes = getattr(file_set, "includes", None) or [] 

130 excludes = getattr(file_set, "excludes", None) or [] 

131 resolved_excludes, excluded = _resolve_patterns( 

132 excludes, by_logical, stored_paths 

133 ) 

134 for entry in _dependants_of(excluded, entries): 

135 resolved_excludes.append(glob.escape(str(entry.path))) 

136 excluded.append(str(entry.path)) 

137 excluded = set(excluded) 

138 

139 resolved, members = _resolve_patterns( 

140 includes, by_logical, stored_paths, excluded 

141 ) 

142 

143 for entry in _dependants_of(members, entries): 

144 if str(entry.path) not in excluded: 144 ↛ 143line 144 didn't jump to line 143 because the condition on line 144 was always true

145 resolved.append(glob.escape(str(entry.path))) 

146 members.append(str(entry.path)) 

147 

148 formats = list(file_set.encoding_formats or []) 

149 file_set.encoding_formats = formats + [ 

150 c.media_type 

151 for c in _wrappers_among(members) 

152 if c.media_type not in formats 

153 ] 

154 if includes: 154 ↛ 156line 154 didn't jump to line 156 because the condition on line 154 was always true

155 file_set.includes = list(dict.fromkeys(resolved)) 

156 if excludes: 

157 file_set.excludes = list(dict.fromkeys(resolved_excludes)) 

158 return file_sets

161def _resolve_patterns( 

162 patterns: list, 

163 by_logical: dict, 

164 stored_paths: dict, 

165 excluded: set | frozenset = frozenset(), 

166) -> tuple: 

167 """Translate logical paths/globs to stored patterns and their members.""" 

168 resolved, members = [], [] 

169 for pattern in patterns: 

170 # An observed filename is exact even if it contains glob characters, 

171 # e.g. slide[1].ome.tif. Escape those characters in the emitted pattern. 

172 is_glob = pattern not in stored_paths and any( 

173 char in pattern for char in _GLOB_CHARS 

174 ) 

175 if is_glob: 

176 matched = [ 

177 str(entry.path) 

178 for logical, found in by_logical.items() 

179 if _matches(pattern, logical) 

180 for entry in found 

181 if str(entry.path) not in excluded 

182 ] 

183 resolved.extend( 

184 compression.expand_globs([pattern], _wrappers_among(matched)) 

185 ) 

186 else: 

187 # Sorted by path: stored forms are recorded in rglob order. 

188 matched = sorted( 

189 p for p in stored_paths.get(pattern, ()) if p not in excluded 

190 ) 

191 resolved.extend(glob.escape(p) for p in matched) 

192 members.extend(matched) 

193 return resolved, members

196def _dependants_of(members: List[str], entries: list) -> list: 

197 """Entries linking to a member, that no pattern already covers. 

198 

199 A duplicate gave up its structure to the file it links to, so it belongs 

200 wherever that file belongs. Named by its stored path, since the name it 

201 duplicates is by definition not the name it is stored under. 

202 """ 

203 covered = set(members) 

204 # Sorted by path: entries arrive in rglob order. 

205 return sorted( 

206 ( 

207 entry 

208 for entry in entries 

209 if entry.duplicate_of is not None 

210 and str(entry.duplicate_of.path) in covered 

211 and str(entry.path) not in covered 

212 ), 

213 key=lambda entry: str(entry.path), 

214 )
